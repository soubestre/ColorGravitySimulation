# Fractal Explorer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a Mandelbrot/Julia WebGPU explorer in `fractals.html` matching the Color Gravity / Galaxies UI, with draft/final quality so interaction stays fluid.

**Architecture:** Single HTML file. CPU holds view + params; GPU compute writes smooth escape to an `r32float` texture; fragment blit colorizes. Interaction renders at reduced resolution; idle promotes to full res. Iterate stage stays isolatable for future deep zoom.

**Tech Stack:** Vanilla JS, WebGPU (WGSL compute + render), same CSS tokens as `galaxies.html`. No build step, no npm.

## Global Constraints

- WebGPU only; French fallback: `WebGPU indisponible — utilisez une version récente de Chrome, Brave, Edge ou Safari.`
- UI language/style matches `galaxies.html` (dark panel, control groups, `#viewPanel` zoom).
- v1 formulas: Mandelbrot + Julia only (no Burning Ship / Newton).
- Float32 only; soft precision warning when scale is extreme — no perturbation.
- Escape work on GPU; CPU uploads uniforms + handles input only.
- Coalesce draws to one in-flight frame via `requestAnimationFrame`.
- Do not commit unrelated dirty files (`index.html` physics edits, etc.) unless the task explicitly modifies nav links.

### Task 2: View math + quality controller (pure JS)

**Files:**
- Modify: `fractals.html` (script section)

**Interfaces:**
- Consumes: canvas/stage DOM from Task 1
- Produces:
  - `view = { cx, cy, scale }` — `scale` = half-height of view in complex plane
  - `screenToComplex(sx, sy, W, H, view) → {x,y}`
  - `zoomAt(view, sx, sy, W, H, factor) → mutates view`
  - `panBy(view, dsx, dsy, W, H) → mutates view`
  - `zoomSliderToScale(t01) → scale` and inverse (`t` in 0..1 maps log scale from ~3.0 down to ~1e-6`)
  - `quality = { interacting, draftScale: 0.25, draftIterCap: 128, idleMs: 100 }`
  - `getRenderBudget(quality, userMaxIter, canvasW, canvasH) → { rw, rh, maxIter, draft: boolean }`
  - `PRECISION_SCALE = 1e-5` — when `view.scale < PRECISION_SCALE`, show warning

- [ ] **Step 1: Implement view helpers**

```js
const view = { cx: -0.5, cy: 0, scale: 1.2 };
const SCALE_MAX = 3.0;
const SCALE_MIN = 1e-6;
const PRECISION_SCALE = 1e-5;

function screenToComplex(sx, sy, W, H, v) {
  const aspect = W / H;
  const x = v.cx + (sx / W - 0.5) * 2 * v.scale * aspect;
  const y = v.cy + (0.5 - sy / H) * 2 * v.scale;
  return { x, y };
}

function panBy(v, dsx, dsy, W, H) {
  const aspect = W / H;
  v.cx -= (dsx / W) * 2 * v.scale * aspect;
  v.cy += (dsy / H) * 2 * v.scale;
}

function zoomAt(v, sx, sy, W, H, factor) {
  const before = screenToComplex(sx, sy, W, H, v);
  v.scale = Math.min(SCALE_MAX, Math.max(SCALE_MIN, v.scale * factor));
  const after = screenToComplex(sx, sy, W, H, v);
  v.cx += before.x - after.x;
  v.cy += before.y - after.y;
}

function zoomSliderToScale(t) {
  // t: 0..1 → SCALE_MAX .. SCALE_MIN (log)
  return Math.exp(Math.log(SCALE_MAX) + t * (Math.log(SCALE_MIN) - Math.log(SCALE_MAX)));
}
function scaleToZoomSlider(scale) {
  return (Math.log(scale) - Math.log(SCALE_MAX)) / (Math.log(SCALE_MIN) - Math.log(SCALE_MAX));
}
```

```js
function getRenderBudget(q, userMaxIter, canvasW, canvasH) {
  if (q.interacting) {

    const rh = Math.max(1, Math.floor(canvasH * q.draftScale));
```

- [ ] **Step 1: WGSL compute — smooth escape → `r32float` storage texture**

```wgsl
struct Uniforms {
  cx: f32, cy: f32, scale: f32, maxIter: f32,
  cRe: f32, cIm: f32, mode: f32, phase: f32,
  palette: f32, resX: f32, resY: f32, _pad: f32,
};
@group(0) @binding(0) var<uniform> U: Uniforms;
@group(0) @binding(1) var escapeTex: texture_storage_2d<r32float, write>;

@compute @workgroup_size(8, 8)
fn csEscape(@builtin(global_invocation_id) gid: vec3u) {
  let rw = u32(U.resX); let rh = u32(U.resY);
  if (gid.x >= rw || gid.y >= rh) { return; }
  let aspect = U.resX / U.resY;
  let uv = vec2f(
    (f32(gid.x) + 0.5) / U.resX,
    (f32(gid.y) + 0.5) / U.resY
  );
  let px = U.cx + (uv.x - 0.5) * 2.0 * U.scale * aspect;
  let py = U.cy + (0.5 - uv.y) * 2.0 * U.scale;

  var zx: f32; var zy: f32; var cx: f32; var cy: f32;
  if (U.mode < 0.5) {
    zx = 0.0; zy = 0.0; cx = px; cy = py; // Mandelbrot
  } else {
    zx = px; zy = py; cx = U.cRe; cy = U.cIm; // Julia
  }

  var i: u32 = 0u;
  let maxI = u32(U.maxIter);
  var escaped = false;
  loop {
    if (i >= maxI) { break; }
    let x2 = zx * zx; let y2 = zy * zy;
    if (x2 + y2 > 256.0) { escaped = true; break; }
    zy = 2.0 * zx * zy + cy;
    zx = x2 - y2 + cx;
    i = i + 1u;
  }

  var t = f32(maxI); // interior
  if (escaped) {
    // smooth escape
    let logZn = log(zx * zx + zy * zy) / 2.0;
    let nu = log(logZn / log(2.0)) / log(2.0);
    t = f32(i) + 1.0 - nu;
  }
  textureStore(escapeTex, vec2i(gid.xy), vec4f(t, 0.0, 0.0, 0.0));
}
```

- [ ] **Step 2: WGSL fragment blit — escape sample → palette**

Fullscreen triangle (or quad) sampling `texture_2d<f32>` (escape as float texture viewed as sampleable — if storage+sample not both allowed on same texture, use: compute writes storage tex A; blit samples via `textureLoad` from a copy, OR use one texture with `TEXTURE_BINDING | STORAGE_BINDING` if the device supports it).

**Preferred path:** create escape texture with  
`usage: STORAGE_BINDING | TEXTURE_BINDING | COPY_SRC`  
If creation fails, fall back to storage-only + `copyTextureToTexture` into a sampleable twin (implement twin only if needed).

Fragment:

```wgsl
@group(0) @binding(0) var escapeTex: texture_2d<f32>;
@group(0) @binding(1) var<uniform> U: Uniforms;

fn paletteColor(t: f32, pal: f32, phase: f32) -> vec3f {
  let x = t * 0.015 + phase;
  if (pal < 0.5) {
    // fire
    return vec3f(
      0.5 + 0.5 * sin(6.28318 * (x + 0.0)),
      0.5 + 0.5 * sin(6.28318 * (x + 0.33)),
      0.2 + 0.3 * sin(6.28318 * (x + 0.67))
    );
  } else if (pal < 1.5) {
    // ocean
    return vec3f(
      0.1 + 0.3 * sin(6.28318 * (x + 0.2)),
      0.3 + 0.4 * sin(6.28318 * (x + 0.4)),
      0.5 + 0.5 * sin(6.28318 * (x + 0.0))
    );
  }
  // mono
  let g = 0.5 + 0.5 * sin(6.28318 * x);
  return vec3f(g, g, g);
}

@fragment
fn fsBlit(@builtin(position) pos: vec4f) -> @location(0) vec4f {
  let coord = vec2i(pos.xy);
  let t = textureLoad(escapeTex, coord, 0).r;
  if (t >= U.maxIter - 0.5) {
    return vec4f(0.02, 0.03, 0.05, 1.0); // interior
  }
  let col = paletteColor(t, U.palette, U.phase);
  return vec4f(col, 1.0);
}
```

```js
const params = {
  mode: 0,
  cRe: -0.4,
  cIm: 0.6,
  maxIter: 256,
  palette: 0,
  phase: 0,
};

let rafPending = false;
let idleTimer = 0;

function requestRender(interactive) {
  if (interactive) quality.interacting = true;
  if (rafPending) return;
  rafPending = true;
  requestAnimationFrame(() => {
    rafPending = false;
    const W = canvas.width, H = canvas.height;
    const budget = getRenderBudget(quality, params.maxIter, W, H);
    renderFrame({
      ...budget,
      cx: view.cx, cy: view.cy, scale: view.scale,
      mode: params.mode, cRe: params.cRe, cIm: params.cIm,
      palette: params.palette, phase: params.phase,
      canvasW: W, canvasH: H,
    });
    document.getElementById("qualityOut").textContent =
      budget.draft ? "draft" : "final";
    // precision warning
    const warn = document.getElementById("warnOut");
    if (view.scale < PRECISION_SCALE) {
      warn.style.display = "";
      warn.textContent = "Précision float32 limite — le détail peut se dégrader.";
    } else {
      warn.style.display = "none";
    }
    clearTimeout(idleTimer);
    if (quality.interacting) {
      idleTimer = setTimeout(() => {
        quality.interacting = false;
        requestRender(false);
      }, quality.idleMs);
    }
  });
}

function resetView() {
  view.cx = -0.5; view.cy = 0; view.scale = 1.2;
  params.mode = 0;
  // sync selects/sliders + zoom slider via scaleToZoomSlider
  requestRender(false);
}
```

```js
const drag = { active: false, moved: false, lx: 0, ly: 0, sx: 0, sy: 0 };

canvas.addEventListener("pointerdown", (e) => {
  canvas.setPointerCapture(e.pointerId);
  const r = canvas.getBoundingClientRect();
  drag.active = true; drag.moved = false;
  drag.lx = e.clientX; drag.ly = e.clientY;
  drag.sx = e.clientX - r.left; drag.sy = e.clientY - r.top;
});

canvas.addEventListener("pointermove", (e) => {
  if (!drag.active) return;
  const dx = e.clientX - drag.lx, dy = e.clientY - drag.ly;
  if (Math.hypot(dx, dy) > 3) drag.moved = true;
  if (!drag.moved) return;
  drag.lx = e.clientX; drag.ly = e.clientY;
  panBy(view, dx, dy, canvas.width, canvas.height);
  // keep zoom slider in sync
  document.getElementById("zoom").value = Math.round(scaleToZoomSlider(view.scale) * 100);
  requestRender(true);
});

canvas.addEventListener("pointerup", (e) => {
  if (!drag.active) return;
  drag.active = false;
  if (!drag.moved && params.mode === 0) {
    const r = canvas.getBoundingClientRect();
    const sx = e.clientX - r.left, sy = e.clientY - r.top;
    // map CSS pixels → canvas buffer pixels
    const x = sx * (canvas.width / r.width);
    const y = sy * (canvas.height / r.height);
    const p = screenToComplex(x, y, canvas.width, canvas.height, view);
    params.mode = 1;
    params.cRe = p.x; params.cIm = p.y;
    document.getElementById("mode").value = "1";
    document.getElementById("cRe").value = String(p.x);
    document.getElementById("cIm").value = String(p.y);
    // update outputs…
    requestRender(true);
  }
});
```
